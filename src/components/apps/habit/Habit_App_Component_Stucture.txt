<script>
  import { actions } from 'astro:actions';

  /**
   * Returns the currently selected tracking_type value from the RadioGroup.
   * Falls back to 'boolean' if nothing is selected.
   *
   * @returns {"boolean" | "count" | "duration"} The selected tracking type.
   */
  function getSelectedTrackingType(): "boolean" | "count" | "duration" {
    const checked = document.querySelector<HTMLInputElement>(
      'input[name="tracking_type"]:checked'
    );
    const value = checked?.value ?? 'boolean';
    if (value === 'count' || value === 'duration') return value;
    return 'boolean';
  }

  /**
   * Resets the form fields and clears any error messages.
   */
  function resetForm() {
    const nameInput = document.getElementById('habit-name') as HTMLInputElement;
    const errorMsg = document.getElementById('habit-form-error') as HTMLElement;
    const nameError = document.getElementById('habit-name-error') as HTMLElement;
    const booleanRadio = document.getElementById('type-boolean') as HTMLInputElement;

    if (nameInput) nameInput.value = '';
    if (booleanRadio) booleanRadio.checked = true;
    if (errorMsg) { errorMsg.textContent = ''; errorMsg.classList.add('hidden'); }
    if (nameError) { nameError.textContent = ''; nameError.classList.add('hidden'); }
  }

  /**
   * Sets the submit button into loading or idle state.
   *
   * @param loading - Whether to show the loading state.
   */
  function setLoading(loading: boolean) {
    const submitBtn = document.getElementById('habit-form-submit') as HTMLButtonElement;
    const label = document.getElementById('habit-submit-label') as HTMLElement;
    const spinner = document.getElementById('habit-submit-spinner') as HTMLElement;

    if (!submitBtn) return;
    submitBtn.disabled = loading;
    label?.classList.toggle('hidden', loading);
    spinner?.classList.toggle('hidden', !loading);
  }

  const submitBtn = document.getElementById('habit-form-submit');

  submitBtn?.addEventListener('click', async () => {
    const nameInput = document.getElementById('habit-name') as HTMLInputElement;
    const errorMsg = document.getElementById('habit-form-error') as HTMLElement;
    const nameError = document.getElementById('habit-name-error') as HTMLElement;

    // Clear previous errors
    errorMsg?.classList.add('hidden');
    nameError?.classList.add('hidden');

    const name = nameInput?.value.trim();

    // Client-side validation
    if (!name) {
      if (nameError) {
        nameError.textContent = 'Habit name is required.';
        nameError.classList.remove('hidden');
      }
      nameInput?.focus();
      return;
    }

    setLoading(true);

    const { data, error } = await actions.habit.create({
      name,
      tracking_type: getSelectedTrackingType(),
    });

    setLoading(false);

    if (error) {
      if (errorMsg) {
        errorMsg.textContent = error.message ?? 'Something went wrong. Please try again.';
        errorMsg.classList.remove('hidden');
      }
      return;
    }

    // Success — close the sheet, reset the form, reload the grid
    document.getElementById('habit-form-cancel')?.click();
    resetForm();

    // Reload the page to show the new habit in the grid
    window.location.reload();
  });
</script>

<script>
  // Extend the global Window interface
declare global {
  interface Window {
    showStreakBanner: (habitName: string, streak: number) => void;

// Define interfaces
interface Particle {
  x: number;
  y: number;
  vx: number;
  vy: number;
  radius: number;
  color: string;
  alpha: number;
  decay: number;
}

interface BannerCall {
  habitName: string;
  streak: number;
}

interface MilestoneCopy {
  emoji: string;
  headline: string;
  sub: string;
}

interface ParticleBurst {
  canvas: HTMLCanvasElement;
  ctx: CanvasRenderingContext2D;
  particles: Particle[];
  runParticleBurst(canvas: HTMLCanvasElement): void;
}

interface StreakBanner {
  showStreakBanner(habitName: string, streak: number): void;
}

interface StreakBannerConstructor {
  new (habitName: string, streak: number): StreakBanner;
}

// Function to get milestone copy
function getMilestoneCopy(streak: number, habitName: string): MilestoneCopy {
  if (streak >= 100) return { emoji: '🏆', headline: `${streak}-day streak on ${habitName}!`, sub: "Absolutely legendary. You've made this part of who you are." };
  if (streak >= 50) return { emoji: '💎', headline: `${streak} days straight — ${habitName}!`, sub: "That's elite-level consistency. Keep going." };
  if (streak >= 30) return { emoji: '⚡', headline: `${streak}-day streak! ${habitName}!`, sub: "A whole month without breaking the chain. Remarkable." };
  if (streak >= 20) return { emoji: '🚀', headline: `${streak} days in a row — ${habitName}!`, sub: "You've built real momentum. Don't stop now." };
  if (streak >= 10) return { emoji: '🔥', headline: `${streak}-day streak on ${habitName}!`, sub: "Double digits. That's a proper habit forming." };
  if (streak >= 5) return { emoji: '⭐', headline: `5-day streak — ${habitName}!`, sub: "Five days straight. You're building something real." };
  return { emoji: '✨', headline: `3-day streak on ${habitName}!`, sub: "Three in a row — the streak has begun." };
}

// Particle burst animation
function runParticleBurst(canvas: HTMLCanvasElement): void {
  const ctx = canvas.getContext('2d');
  if (!ctx) return;

  canvas.width = canvas.offsetWidth;
  canvas.height = canvas.offsetHeight;

  const colors = ['#fbbf24', '#f472b6', '#34d399', '#60a5fa', '#a78bfa', '#fff'];
  const particles: Particle[] = [];

  for (let i = 0; i < 60; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 1.5 + Math.random() * 3.5;
    particles.push({
      x: canvas.width * (0.2 + Math.random() * 0.6),
      y: canvas.height / 2,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed - 2,
      radius: 2 + Math.random() * 3,
      color: colors[Math.floor(Math.random() * colors.length)],
      alpha: 1,
      decay: 0.018 + Math.random() * 0.012,
    });
  }

  function frame() {
    ctx!.clearRect(0, 0, canvas.width, canvas.height);
    let alive = false;
    for (const p of particles) {
      p.x += p.vx;
      p.y += p.vy;
      p.vy += 0.12;
      p.alpha -= p.decay;
      if (p.alpha > 0) {
        alive = true;
        ctx!.save();
        ctx!.globalAlpha = p.alpha;
        ctx!.fillStyle = p.color;
        ctx!.beginPath();
        ctx!.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
        ctx!.fill();
        ctx!.restore();
      }
    }
    if (alive) {
      requestAnimationFrame(frame);
    }
  }

  requestAnimationFrame(frame);
}

// Queue for pending calls
const pendingCalls: BannerCall[] = [];

/**
 * Internal banner display implementation.
 * Assigned after DOMContentLoaded.
 */
let _showBanner:
  | ((habitName: string, streak: number) => void)
  | null = null;

/**
 * Public global API.
 */
window.showStreakBanner = function (
  habitName: string,
  streak: number
): void {
  if (_showBanner) {
    _showBanner(habitName, streak);
  } else {
    pendingCalls.push({ habitName, streak });
  }
};

// DOM wiring
document.addEventListener('DOMContentLoaded', () => {
  const banner = document.getElementById('streak-banner');
  const emojiEl = document.getElementById('streak-banner-emoji');
  const headlineEl = document.getElementById('streak-banner-headline');
  const subEl = document.getElementById('streak-banner-sub');
  const closeBtn = document.getElementById('streak-banner-close');
  const canvas = document.getElementById('streak-canvas');

  let dismissTimer: number | null = null;

  // Define the actual implementation of showStreakBanner
  _showBanner = function(habitName: string, streak: number): void {
    if (!banner || !emojiEl || !headlineEl || !subEl || !closeBtn || !canvas) return;

    if (dismissTimer) {
      clearTimeout(dismissTimer);
    }

    const copy = getMilestoneCopy(streak, habitName);
    emojiEl.textContent = copy.emoji;
    headlineEl.textContent = copy.headline;
    subEl.textContent = copy.sub;

    banner.classList.add('is-visible');
    // Wait for transition
    setTimeout(() => {
      runParticleBurst(canvas as HTMLCanvasElement);
    }, 420);
    dismissTimer = window.setTimeout(() => {
      banner.classList.remove('is-visible');
    }, 5000);
  };

  // Close button event
  if (closeBtn) {
    closeBtn.addEventListener('click', () => {
      if (banner) banner.classList.remove('is-visible');
      if (dismissTimer) {
        clearTimeout(dismissTimer);
        dismissTimer = null;
      }
    });
  }

  // Process any calls made before DOMContentLoaded
  for (const call of pendingCalls) {
    _showBanner?.(call.habitName, call.streak);
  }
  pendingCalls.length = 0;
});

export {}; // To ensure this is a module
</script>

<script>
  import { actions } from 'astro:actions';
  import { toast } from '@/components/starwind/toast';

  /**
   * Handles clicking a habit cell to toggle completion for that day.
   *
   * Optimistic UI — the cell updates immediately. The action runs in the
   * background. On failure the cell reverts and a toast error is shown.
   *
   * On success, if the action returns a streak milestone, we call
   * window.showStreakBanner() which is registered by StreakBanner.astro.
   */
  document.querySelectorAll<HTMLElement>('.habit-cell').forEach(cell => {
    cell.addEventListener('click', async () => {
      const habitId   = cell.getAttribute('data-habit-id')!;
      const habitName = cell.getAttribute('data-habit-name')!;
      const date      = cell.getAttribute('data-date')!;
      const isCompleted = cell.classList.contains('is-completed');

      if (isCompleted) {
        const logId = cell.getAttribute('data-log-id')!;

        // Optimistic: remove immediately
        cell.classList.remove('is-completed');
        const prevLogId = logId;
        cell.setAttribute('data-log-id', '');

        const { error } = await actions.habit.deleteLog({
          id: logId,
          habit_id: habitId,
          entry_date: date,
        });

        if (error) {
          // Rollback
          cell.classList.add('is-completed');
          cell.setAttribute('data-log-id', prevLogId);
          toast.error('Could not remove log', { description: 'Please try again.' });
        }

      } else {
        // Optimistic: mark done immediately
        cell.classList.add('is-completed');

        const { data, error } = await actions.habit.upsertLog({
          habit_id: habitId,
          entry_date: date,
          value: 1,
        });

        if (error) {
          // Rollback
          cell.classList.remove('is-completed');
          toast.error('Could not save log', { description: 'Please try again.' });
          return;
        }

        // Store the real log id for future delete
        if (data) {
          cell.setAttribute('data-log-id', data.id);
        }

        // Show streak milestone banner if one was hit
        const streak = data?.streak;
        if (streak?.milestone) {
          const showBanner = (window as any).showStreakBanner;
          if (typeof showBanner === 'function') {
            showBanner(habitName, streak.currentStreak);
          }
        }
      }
    });
  });
</script>
