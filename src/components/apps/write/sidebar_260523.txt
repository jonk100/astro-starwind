<script>

  /**───────────────────── Folder toggle ───────────────────────────────────────
   * Toggles a folder section open or closed when the folder header is clicked.
   * Updates aria-expanded so assistive technology is notified of state.
   * @param btn - The folder toggle button element.
   */
  function initFolderToggles() {
    document.querySelectorAll<HTMLButtonElement>("[data-folder-toggle]").forEach((btn) => {
      btn.addEventListener("click", () => {
        const expanded = btn.getAttribute("aria-expanded") === "true";
        const targetId = btn.getAttribute("aria-controls");
        const target = targetId ? document.getElementById(targetId) : null;
        btn.setAttribute("aria-expanded", String(!expanded));
        if (target) {
          target.style.display = expanded ? "none" : "";
        }
      });
    });
  }

  /**───────────────────── Live Updates───────────────────────────────────
   * Listens for client-side events fired by the editor canvas and updates
   * the sidebar text nodes in real-time without causing a page re-render.
   */
  function initRealtimeUpdates() {

    // 1. Sync Title changes smoothly as the user types
    window.addEventListener("journal:title-updated", ((e: CustomEvent) => {
      const { documentId, title } = e.detail;      
      const docItems = document.querySelectorAll(`[data-doc-id="${documentId}"]`);
      docItems.forEach((item) => {
        const titleEl = item.querySelector(".doc-title");
        if (titleEl) titleEl.textContent = title;
      });
    }) as EventListener);

    // 2. Sync Meta timings ("Today", "Yesterday") when autosave fires successfully
    window.addEventListener("journal:doc-updated", ((e: CustomEvent) => {
      const { documentId, updatedAt } = e.detail;
      const docItems = document.querySelectorAll(`[data-doc-id="${documentId}"]`);
      docItems.forEach((item) => {
        const metaEl = item.querySelector(".doc-meta");
        if (metaEl) {
          // Quick simplified runtime client format mirror
          metaEl.textContent = "Today";
        }
      });
    }) as EventListener);

    // 3. Sync Pinned status instantly______________________________________________
    window.addEventListener("journal:doc-pinned", ((e: CustomEvent) => {
      const { documentId, pinned } = e.detail;
      const docItems = document.querySelectorAll(`[data-doc-id="${documentId}"]`);
      
      docItems.forEach((item) => {
        const list = item.parentElement;
        if (!list) return;

        if (pinned) {
          // Visually mark it (optional, if you want to add a pin icon in the sidebar via CSS)
          item.classList.add("is-pinned");
          // Move the DOM element to the very top of its parent <ul> list!
          list.prepend(item);
        } else {
          item.classList.remove("is-pinned");
          // Note: When unpinned, it stays at the top until the page is refreshed. 
          // Dynamically re-sorting by date in the DOM is heavy, so leaving it 
          // there temporarily is standard practice for optimistic UIs.
        }
      });
    }) as EventListener);
  }
  

  // ── Init ──────────────────────────────────────────
  document.addEventListener("DOMContentLoaded", () => {
    initFolderToggles();
    initRealtimeUpdates(); 
  });
</script>

<script>
  import { actions } from "astro:actions";
  /**
   * Renders a list of search result items into the search results panel.
   * @param results - Array of result objects with id, title, preview, updated_at.
   */
  function renderSearchResults(results: Array<{
    id: string;
    title: string;
    preview: string | null;
    updated_at: string;
  }>) {
    const list = document.getElementById("sidebar-search-results");
    if (!list) return;

    if (results.length === 0) {
      list.innerHTML = `<li class="doc-empty" style="padding: 1rem;">No results found.</li>`;
      return;
    }

    list.innerHTML = results
      .map(
        (doc) => `
          <li class="doc-item">
            <a href="/app/write?doc=${doc.id}" class="doc-link">
              <span class="doc-title">${doc.title || "Untitled"}</span>
              ${doc.preview ? `<span class="doc-meta">${doc.preview.slice(0, 60)}…</span>` : ""}
            </a>
          </li>
        `
      )
      .join("");
  }

  /**
   * Wires up the search input with a debounced call to journal.search.
   * Shows the search results overlay while a query is active,
   * and hides it when the input is cleared.
   */
  function initSearch() {
    const input = document.getElementById("sidebar-search-input") as HTMLInputElement | null;
    const results = document.getElementById("sidebar-search-results");
    const body = document.getElementById("sidebar-document-list");
    if (!input || !results || !body) return;

    let debounceTimer: ReturnType<typeof setTimeout>;

    input.addEventListener("input", () => {
      clearTimeout(debounceTimer);
      const query = input.value.trim();

      if (!query) {
        // Clear search — show normal document list
        results.classList.add("hidden");
        body.style.display = "";
        return;
      }

      // * Show results overlay, hide normal list
      results.classList.remove("hidden");
      body.style.display = "none";
      debounceTimer = setTimeout(async () => {
        const { data, error } = await actions.write.search({ query });
        if (error) {
          console.error("Search failed:", error.message);
          return;
        }
        renderSearchResults(data ?? []);
      }, 300);
    });
  }
  
  // ── Init ───────────────────────────────────────────────────────────────────
  document.addEventListener("DOMContentLoaded", () => {
    initSearch();
  });
</script>

import type { DocumentIndexRow } from "@/lib/journal/types";

//! - the function takes 'documents' as an argument
export function sortDocumentsForSidebar(documents: DocumentIndexRow[]) {

  //! 1. Define comparison, get the date and pinned state
  function compareDocuments(a: DocumentIndexRow, b: DocumentIndexRow) {
    if (a.pinned === b.pinned) {
      return new Date(b.updated_at).getTime() - new Date(a.updated_at).getTime();
    }
    return a.pinned ? -1 : 1;
  }

  //! 2. Do the sorting and return the result
  return [...documents].sort(compareDocuments);
}

//! Pass the sortedArguments to this function
export function groupDocumentsByFolder(documents: DocumentIndexRow[]) {
  const byFolder = new Map<string | null, DocumentIndexRow[]>();
  for (const doc of documents) {
    const key = doc.folder_id ?? null;
    if (!byFolder.has(key)) {
      byFolder.set(key, []);
    }
    byFolder.get(key)!.push(doc);
  }
  // Grab the ones with no folder
  const unfiledDocs = byFolder.get(null) ?? [];
  //! Return both pieces of data as an object so your Astro file can use them
  return { 
    byFolder: byFolder, 
    unfiledDocs: unfiledDocs 
  };
}

/** 
 *! Format an ISO timestamp into a human-readable relative date label.
 *! - used in the document list items.
 * @param iso - An ISO 8601 date string from the database.
 * @returns A short label like "Today", "Yesterday", or "12 May".
 */
export function formatUpdatedAt(iso: string): string {
  const date = new Date(iso);
  const now = new Date();
  const isToday =
    date.getDate() === now.getDate() &&
    date.getMonth() === now.getMonth() &&
    date.getFullYear() === now.getFullYear();
  if (isToday) return "Today";
  const yesterday = new Date(now);
  yesterday.setDate(now.getDate() - 1);
  const isYesterday =
    date.getDate() === yesterday.getDate() &&
    date.getMonth() === yesterday.getMonth() &&
    date.getFullYear() === yesterday.getFullYear();
  if (isYesterday) return "Yesterday";
  return date.toLocaleDateString("en-GB", { day: "numeric", month: "short" });
}
