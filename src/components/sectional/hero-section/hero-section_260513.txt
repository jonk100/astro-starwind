  textClass?: string;
  as?: keyof HTMLElementTagNameMap;
  imgSrc?: string;
  imgAlt?: string;

  height?: number;
  action?: 'button' | 'search' | 'link';
}

---
<Center axis="y" class="h-[90vh]">
<!-- If image is provided, use it -->
{imgSrc ? 
  <Element class={`${className} mx-auto section-container hero-section`} style={style}>
    <HeroImage src={imgSrc} alt={imgAlt} width={width} height={height} />

    {/* If action is provided, use it */}
    {action && <HeroAction type={action} />}
  </Element>

    <HeroText h={h} p={p} class={textClass} large={!imgSrc} />

  </Element>
