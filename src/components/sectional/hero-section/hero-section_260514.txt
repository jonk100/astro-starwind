import HeroText from './hero-text.astro';
import HeroImage from './hero-image.astro';
import HeroAction from './hero-action.astro';
import type { HTMLAttributes } from 'astro/types';
import Center from '@/components/layout/Center.astro';

/** A button action with an optional link destination. */
interface ActionButton {
  type: 'button';
  text: string;
  href?: string;
}

/** A search/select action with a list of menu items. */
interface ActionSearch {
  type: 'search';
  placeholder: string;
  name: string;
  menuItems: string[];
}

/** A plain anchor link action. */
interface ActionLink {
  type: 'link';
  text: string;
  href: string;
}

/** Union of all supported hero action shapes. */
type HeroActionProps = ActionButton | ActionSearch | ActionLink;

interface Props extends HTMLAttributes<'section'> {
  h: string;
  p: string;
  style?: string;
  class?: string;
  textClass?: string;
  as?: keyof HTMLElementTagNameMap;
  imgSrc?: string;
  imgAlt?: string;
  width?: number;
  height?: number;
  /** Pass the full action object so HeroAction receives all required props. */
  action?: HeroActionProps;
}

const {
  h,
  p,
  style,
  class: className = '',
  textClass = '',
  as: Element = 'section',
  imgSrc,
  imgAlt = '',
  width = 0,
  height = 0,
  action,
} = Astro.props;
---

{imgSrc ? (
  <Element
    class={`${className} relative flex w-full min-h-[90vh] flex-col items-center justify-center overflow-hidden px-4 py-12 hero-section`}
    style={style}
  >
    <HeroImage src={imgSrc} alt={imgAlt} width={width} height={height} />
    <div
      class="hero-gradient"
      aria-hidden="true"
    />
    <div
      class="
        pointer-events-none
        absolute
        inset-0
        z-2
        bg-linear-to-b
        from-black/10
        via-black/30
        to-black/10
      "
      aria-hidden="true"
    />
    <HeroText
      h={h}
      p={p}
      class={`relative z-10 w-3/4 max-w-3xl ${textClass}`}
      tone="onImage"
    />
    {action && <HeroAction {...action} />}
  </Element>
) : (
  <Center axis="y" class="h-[90vh] w-full">
    <Element class={`${className} mx-auto section-container`} style={style}>
      <HeroText h={h} p={p} class={textClass} large />
      {action && <HeroAction {...action} />}
    </Element>
  </Center>
)}

  type: 'button';
  text: string;
  href?: string;
}

interface ActionSearch {
  type: 'search';
  placeholder: string;
  name: string;
  menuItems: string[];
}

interface ActionLink {
  type: 'link';
  text: string;
  href: string;
}

type Props = ActionButton | ActionSearch | ActionLink;

const props = Astro.props;
---

{props.type === 'button' && (
  <Button href={props.href}>{props.text}</Button>
)}

{props.type === 'search' && (
  <Select id="combobox-demo" name={props.name}>
    <SelectTrigger class="w-[240px]">
      <SelectValue placeholder={props.placeholder} />
    </SelectTrigger>
    <SelectContent>
      <SelectSearch placeholder={`Search ${props.name}...`} emptyText="No results found." />
      <SelectGroup>
        {props.menuItems.map((item) => (
          <SelectItem value={item}>{item}</SelectItem>
        ))}
      </SelectGroup>
    </SelectContent>
  </Select>
)}

{props.type === 'link' && (
  <a href={props.href}>{props.text}</a>
