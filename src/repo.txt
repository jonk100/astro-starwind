   * Updates the save status indicator in the toolbar.
   *
   * The editor canvas dispatches a custom "editor:save-status" event
   * with a detail of "saving", "saved", or "error". This listener
   * reflects that state in the toolbar text and data-status attribute.
   *
   * @param status - The current save state.
   * @param message - Optional override message to display.
   */
  function setSaveStatus(status: "saving" | "saved" | "error", message?: string) {
    const el = document.getElementById("save-status");
    if (!el) return;

    const labels: Record<string, string> = {
      saving: "Saving…",
      saved: "Saved",
      error: "Save failed",
    };

    el.setAttribute("data-status", status);
    el.textContent = message ?? labels[status] ?? status;
  }

  window.addEventListener("editor:save-status", (e: Event) => {
    const detail = (e as CustomEvent<{ status: "saving" | "saved" | "error" }>).detail;
    setSaveStatus(detail.status);
  });

</script>

  function initDeleteButton() {

        btn.removeAttribute("disabled");
        console.error("Failed to delete document:", error.message);
        return;

    const btn = document.getElementById("toolbar--pin-button") as HTMLButtonElement | null;
    if (!btn) return;

    const documentId = btn.getAttribute("data-document-id");
    if (!documentId) return;

      const newPinned = !currentlyPinned;

      // Optimistic UI Update (This changes the color instantly!)
      btn.setAttribute("aria-pressed", String(newPinned));

      // Broadcast the event to the sidebar
      window.dispatchEvent(new CustomEvent("journal:doc-pinned", {
        detail: { documentId, pinned: newPinned }

        id: documentId,
        pinned: newPinned,
      });

      if (error) {
        // Rollback on failure
        btn.setAttribute("aria-pressed", String(currentlyPinned));
        window.dispatchEvent(new CustomEvent("journal:doc-pinned", {
          detail: { documentId, pinned: currentlyPinned }
        }));
        console.error("Failed to pin document:", error.message);
      }
    });
  }

  document.addEventListener("DOMContentLoaded", () => {
    initPinButton();

/**
 * @file src/components/apps/write/editor/editorCanvas.ts
 * @description Client-side editor logic for the Write app.
 * 
 * Handles contenteditable block editing, keyboard shortcuts, block transformations,
 * and debounced autosave to Supabase via Astro actions.
 * 
 * This module is initialized once per document edit session by the
 * EditorCanvas.astro component, which passes the document ID through
 * a data attribute on the canvas element.
 */

import type { Block, EditorState, BlockType } from "@/lib/journal/types";
import { actions } from "astro:actions";
import { debouncedSaveTitle } from "./utils"; // Verify this points to your file

// ─── Constants & Configuration ──────────────────────────────────────────────
const AUTOSAVE_DEBOUNCE_MS = 250;

// ─── State Management ──────────────────────────────────────────────────────
const documentStates = new Map<string, EditorState>();

function updateEditorState(documentId: string, updater: (state: EditorState) => EditorState) {
  const currentState = documentStates.get(documentId);
  const newState = updater(currentState ?? {
    documentId,
    blocks: [],
    isDirty: false,
    isSaving: false,
    lastSavedAt: null,
  });

  documentStates.set(documentId, newState);
  updateSavingIndicator(documentId, newState.isSaving);
  updateLastSavedTime(documentId, newState.lastSavedAt);
}

// ─── DOM Helpers ───────────────────────────────────────────────────────────
function getCanvasElement(documentId: string): HTMLElement | null {
  return document.querySelector(`#editor-canvas[data-document-id="${documentId}"]`);
}

function getTitleElement(documentId: string): HTMLElement | null {
  const canvas = getCanvasElement(documentId);
  if (!canvas) return null;
  return canvas.querySelector(".editor-title");
}

function getBlockElements(documentId: string): NodeListOf<HTMLElement> {
  const canvas = getCanvasElement(documentId);
  if (canvas) {
    return canvas.querySelectorAll(".editor-block");
  }
  return document.querySelectorAll(".editor-block") as unknown as NodeListOf<HTMLElement>;
}

function updateSavingIndicator(documentId: string, isSaving: boolean) {
  const canvas = getCanvasElement(documentId);
  if (!canvas) return;
  const indicator = canvas.querySelector(".saving-indicator") as HTMLElement;
  if (indicator) {
    indicator.style.opacity = isSaving ? "1" : "0";
  }
}

function updateLastSavedTime(documentId: string, lastSavedAt: string | null) {
  const canvas = getCanvasElement(documentId);
  if (!canvas) return;
  const timeDisplay = canvas.querySelector(".last-saved-time");
  if (timeDisplay && lastSavedAt) {
    const date = new Date(lastSavedAt);
    timeDisplay.textContent = date.toLocaleTimeString();
  }
}

// ─── Block Management ──────────────────────────────────────────────────────
// Fixed fallback string checking syntax rule safely for runtime
function getBlockId(element: HTMLElement): string | null {
  return element.dataset.blockId || element.getAttribute("data-block-id") || null;
}

function getBlockType(element: HTMLElement): Block["type"] {
  const type = element.dataset.blockType || element.getAttribute("data-block-type");
  return (type as Block["type"]) || "paragraph";
}

function createBlockElement(block: Block): HTMLElement {
  const Tag = block.type === "heading" ? "h2" : "div";
  const className = block.type === "heading" ? "editor-block editor-block--heading" : `editor-block editor-block--${block.type}`;

  const el = document.createElement(Tag);
  el.className = className;
  el.contentEditable = block.type !== "separator" ? "true" : "false";
  el.setAttribute("data-block-id", block.id);
  el.setAttribute("data-block-type", block.type);
  el.textContent = block.content;
  return el;
}

/**
 * Parses the current block list from the DOM.
 */
function parseBlocksFromDOM(documentId: string): Block[] {
  const elements = getBlockElements(documentId);
  if (!elements) return [];

  const blocks: Block[] = [];
  Array.from(elements).forEach((el) => {
    // Crucial validation: Skip empty default paragraph states if they don't have structural markers yet
    const id = getBlockId(el) || generateBlockId();
    const type = getBlockType(el);
    blocks.push({
      id,
      type,
      content: el.textContent || "",
      meta: el.dataset.meta ? JSON.parse(el.dataset.meta) : undefined,
    });
  });

  return blocks;
}

function generateBlockId(): string {
  return `b_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`;
}

// ─── Autosave & Actions ────────────────────────────────────────────────────

/**
 * Saves the current blocks to the server via the Astro action.
 */
async function saveBlocks(documentId: string, blocks: Block[]) {
  const state = documentStates.get(documentId);
  if (!state) return;

  updateEditorState(documentId, (prev) => ({ ...prev, isSaving: true }));

  try {
    // Calls write action securely via client proxy reference architecture
    const { data, error } = await actions.write.saveBlocks({
      document_id: documentId,
      blocks,
    });

    // Broadcast that the document updated its content/meta
    window.dispatchEvent(new CustomEvent("journal:doc-updated", {
      detail: { 
        documentId, 
        updatedAt: data?.updated_at || new Date().toISOString() 
      }
    }));

    if (error) throw new Error(`Save failed: ${error.message}`);

    updateEditorState(documentId, (prev) => ({
      ...prev,
      isDirty: false,
      isSaving: false,
      lastSavedAt: data?.updated_at || new Date().toISOString(),
      blocks,
    }));
  } catch (error) {
    console.error("Autosave error:", error);
    updateEditorState(documentId, (prev) => ({ ...prev, isSaving: false }));
  }
}

let saveTimeout: ReturnType<typeof setTimeout> | null = null;
function debouncedSave(documentId: string, blocks: Block[]) {
  if (saveTimeout) clearTimeout(saveTimeout);

  saveTimeout = setTimeout(() => {
    saveBlocks(documentId, blocks);
    saveTimeout = null;
  }, AUTOSAVE_DEBOUNCE_MS);
}

function handleBlockInput(event: Event) {
  const target = event.target as HTMLElement;
  if (!target.classList.contains("editor-block") && !target.classList.contains("editor-title")) return;

  const element = target.closest("#editor-canvas");
  if (!(element instanceof HTMLElement)) return;

  const documentId = element.dataset.documentId;
  if (!documentId) return;

  if (target.classList.contains("editor-title")) {
    const newTitle = target.textContent || "";
    debouncedSaveTitle(documentId, newTitle);

    // Add this: Instantly broadcast title mutations as they happen!
    window.dispatchEvent(new CustomEvent("journal:title-updated", {
      detail: { documentId, title: newTitle || "Untitled" }
    }));
    return;
  }

  updateEditorState(documentId, (prev) => ({ ...prev, isDirty: true }));
  const blocks = parseBlocksFromDOM(documentId);
  debouncedSave(documentId, blocks);
}

/**
 * Handles keyboard events for special actions.
 */
function handleBlockKeyDown(event: KeyboardEvent) {
  const target = event.target as HTMLElement;
  const isBlock = target.classList.contains("editor-block");
  const isTitle = target.classList.contains("editor-title");
  if (!isBlock && !isTitle) return;

  const element = target.closest("#editor-canvas");
  if (!(element instanceof HTMLElement)) return;

  const documentId = element.dataset.documentId;
  if (!documentId) return;

  // Fix Enter mechanics inside Document Title
  if (event.key === "Enter" && !event.shiftKey) {
    event.preventDefault();

    if (isTitle) {
      // Shift down focus automatically into the first block element on submit
      const blocksList = element.querySelectorAll(".editor-block");
      if (blocksList.length > 0) {
        (blocksList[0] as HTMLElement).focus();
      }
      return;
    }

    // Inside normal writing block sequence
    const blocks = parseBlocksFromDOM(documentId);
    const blockElements = Array.from(getBlockElements(documentId) || []);
    const blockIndex = blockElements.findIndex((el) => el === target);

    const newBlock: Block = {
      id: generateBlockId(),
      type: "paragraph" as BlockType,
      content: "",
    };

    const newEl = createBlockElement(newBlock);
    target.after(newEl);
    newEl.focus();

    const updatedBlocks = [...blocks];
    updatedBlocks.splice(blockIndex + 1, 0, newBlock);
    updateEditorState(documentId, (prev) => ({ ...prev, isDirty: true, blocks: updatedBlocks }));
    debouncedSave(documentId, updatedBlocks);
    return;
  }

  // Handle up, down and backward clean text collapses
  if (isTitle) return; // Arrow mapping routines are bound only to workspace textblocks

  const blocks = parseBlocksFromDOM(documentId);
  const blockElements = Array.from(getBlockElements(documentId) || []);
  const blockIndex = blockElements.findIndex((el) => el === target);
  const currentBlock = blocks[blockIndex];

  if (event.key === "Backspace") {
    if (target.textContent?.length === 0 && blocks.length > 1) {
      event.preventDefault();
      const prevBlock = blocks[blockIndex - 1];
      target.remove();

      if (prevBlock) {
        const prevEl = Array.from(getBlockElements(documentId) || []).find(
          (el) => getBlockId(el) === prevBlock.id
        );
        if (prevEl) {
          prevEl.focus();
          const range = document.createRange();
          range.selectNodeContents(prevEl);
          range.collapse(false);
          const selection = window.getSelection();
          selection?.removeAllRanges();
          selection?.addRange(range);
        }
      }

      const updatedBlocks = [...blocks];
      updatedBlocks.splice(blockIndex, 1);
      updateEditorState(documentId, (prev) => ({ ...prev, isDirty: true, blocks: updatedBlocks }));
      debouncedSave(documentId, updatedBlocks);
    }
  }

  if (event.key === "ArrowUp" || event.key === "ArrowDown") {
    const direction = event.key === "ArrowUp" ? -1 : 1;
    const newIndex = blockIndex + direction;

    if (newIndex >= 0 && newIndex < blocks.length) {
      event.preventDefault();
      const targetEl = blockElements[newIndex];
      if (targetEl) {
        targetEl.focus();
        const range = document.createRange();
        range.selectNodeContents(targetEl);
        range.collapse(direction === -1 ? false : true);
        const selection = window.getSelection();
        selection?.removeAllRanges();
        selection?.addRange(range);
      }
    }
  }
}

function handleBlockFocus(event: Event) {
  const target = event.target as HTMLElement;
  if (target.classList.contains("editor-block")) {
    target.setAttribute("data-focused", "true");
  }
}

// Fixed listener attachment signature loop to keep it clean and robust
function handleBlockBlur(event: Event) {
  const target = event.target as HTMLElement;
  if (target.classList.contains("editor-block")) {
    target.removeAttribute("data-focused");
  }
}

// ─── Initialization ────────────────────────────────────────────────────────

/**
 * Clean signature hook alignment.
 */
export function initEditor(documentId: string) {
  const initialBlocks = parseBlocksFromDOM(documentId);
  updateEditorState(documentId, (prev) => ({ ...prev, blocks: initialBlocks }));

  const canvas = getCanvasElement(documentId);
  if (!canvas) return;

  canvas.addEventListener("input", handleBlockInput);
  canvas.addEventListener("keydown", handleBlockKeyDown);
  canvas.addEventListener("focus", handleBlockFocus, true);
  canvas.addEventListener("blur", handleBlockBlur, true);
}

export function disposeEditor(documentId: string) {
  documentStates.delete(documentId);
}

    // 1. Sync Title changes smoothly as the user types
    window.addEventListener("journal:title-updated", ((e: CustomEvent) => {
      const { documentId, title } = e.detail;      
      const docItems = document.querySelectorAll(`[data-doc-id="${documentId}"]`);
      docItems.forEach((item) => {
        const titleEl = item.querySelector(".doc-title");
        if (titleEl) titleEl.textContent = title;

          // Quick simplified runtime client format mirror
          metaEl.textContent = "Today";
        }
      });
    }) as EventListener);

    // 3. Sync Pinned status instantly______________________________________________
    window.addEventListener("journal:doc-pinned", ((e: CustomEvent) => {
      const { documentId, pinned } = e.detail;
      const docItems = document.querySelectorAll(`[data-doc-id="${documentId}"]`);
      
      docItems.forEach((item) => {
        const list = item.parentElement;

import type { DocumentIndexRow } from "@/lib/journal/types";

//! - the function takes 'documents' as an argument
export function sortDocumentsForSidebar(documents: DocumentIndexRow[]) {

  //! 1. Define comparison, get the date and pinned state
  function compareDocuments(a: DocumentIndexRow, b: DocumentIndexRow) {
    if (a.pinned === b.pinned) {
      return new Date(b.updated_at).getTime() - new Date(a.updated_at).getTime();
    }
    return a.pinned ? -1 : 1;
  }

  //! 2. Do the sorting and return the result
  return [...documents].sort(compareDocuments);
}

//! Pass the sortedArguments to this function
export function groupDocumentsByFolder(documents: DocumentIndexRow[]) {
  const byFolder = new Map<string | null, DocumentIndexRow[]>();
  for (const doc of documents) {
    const key = doc.folder_id ?? null;
    if (!byFolder.has(key)) {
      byFolder.set(key, []);
    }
    byFolder.get(key)!.push(doc);
  }
  // Grab the ones with no folder
  const unfiledDocs = byFolder.get(null) ?? [];
  //! Return both pieces of data as an object so your Astro file can use them
  return { 
    byFolder: byFolder, 
    unfiledDocs: unfiledDocs 
  };
}

/** 
 *! Format an ISO timestamp into a human-readable relative date label.
 *! - used in the document list items.
 * @param iso - An ISO 8601 date string from the database.
 * @returns A short label like "Today", "Yesterday", or "12 May".
 */
export function formatUpdatedAt(iso: string): string {
  const date = new Date(iso);
  const now = new Date();
  const isToday =
    date.getDate() === now.getDate() &&
    date.getMonth() === now.getMonth() &&
    date.getFullYear() === now.getFullYear();
  if (isToday) return "Today";
  const yesterday = new Date(now);
  yesterday.setDate(now.getDate() - 1);
  const isYesterday =
    date.getDate() === yesterday.getDate() &&
    date.getMonth() === yesterday.getMonth() &&
    date.getFullYear() === yesterday.getFullYear();
  if (isYesterday) return "Yesterday";
  return date.toLocaleDateString("en-GB", { day: "numeric", month: "short" });
}

// ─── Shared Types ─────────────────────────────────────────────────────────────

/**
 * Standard success/failure response shape returned by all auth actions.
 * The client checks `success` first, then reads `message` for display.
 */
export interface AuthResult {
  success: boolean;
  message: string;
}

// ─── Helpers ──────────────────────────────────────────────────────────────────

/**
 * Maps Supabase error messages to user-friendly equivalents.
 * Supabase's own messages are generally safe to surface, but some are
 * cryptic or too technical for a login form. This normalises the worst ones.
 *
 * @param message - The raw error.message string from Supabase.
 * @returns A user-facing string suitable for display in the form.
 */
export function friendlyAuthError(message: string): string {
  const lower = message.toLowerCase();

  if (lower.includes("invalid login credentials")) {
    return "Incorrect email or password.";
  }
  if (lower.includes("email not confirmed")) {
    return "Please confirm your email before signing in. Check your inbox.";
  }
  if (lower.includes("user already registered")) {
    return "An account with this email already exists. Try signing in.";
  }
  if (lower.includes("password should be at least")) {
    return "Password must be at least 6 characters.";
  }
  if (lower.includes("rate limit")) {
    return "Too many attempts. Please wait a moment and try again.";
  }
  if (lower.includes("network") || lower.includes("fetch")) {
    return "Network error. Check your connection and try again.";
  }

  // Fall back to the raw message — Supabase's defaults are reasonable.
  return message;
}

// ─── Actions ──────────────────────────────────────────────────────────────────

export const auth = {
  /**
   * Signs in an existing user with email and password.
   * Returns a success/failure result rather than throwing, so the
   * client can display inline form errors without a try/catch.
   *
   * @input email - The user's email address.
   * @input password - The user's password.
   */
  signIn: defineAction({
    accept: "form",
    input: z.object({
      email: z.string().email("Please enter a valid email address.").trim(),
      password: z.string().min(1, "Password is required."),
    }),
    handler: async (input, context): Promise<AuthResult> => {
      const supabase = createClient(context);

      const { error } = await supabase.auth.signInWithPassword({
        email: input.email,
        password: input.password,
      });

      if (error) {
        return {
          success: false,
          message: friendlyAuthError(error.message),
        };
      }

      return {
        success: true,
        message: "Signed in successfully.",
      };
    },
  }),

  /**
   * Registers a new user with email and password.
   * Sends a confirmation email — the user must verify before signing in.
   *
   * @input email - The user's email address.
   * @input password - The user's chosen password (min 6 characters).
   */
  signUp: defineAction({
    accept: "form",
    input: z.object({
      email: z.string().email("Please enter a valid email address.").trim(),
      password: z.string().min(6, "Password must be at least 6 characters."),
    }),
    handler: async (input, context): Promise<AuthResult> => {
      const supabase = createClient(context);
      const origin = new URL(context.request.url).origin;
      const emailRedirectTo = `${origin}/auth/callback`;

      const { error } = await supabase.auth.signUp({
        email: input.email,
        password: input.password,
        options: { emailRedirectTo },
      });

      if (error) {
        return {
          success: false,
          message: friendlyAuthError(error.message),
        };
      }

      return {
        success: true,
        message: "Check your email to confirm your account.",
      };
    },
  }),

  /**
   * Signs out the current user and clears the session cookie.
   */
  signOut: defineAction({
    handler: async (_, context): Promise<AuthResult> => {
      const supabase = createClient(context);

      const { error } = await supabase.auth.signOut();

      if (error) {
        return {
          success: false,
          message: "Failed to sign out. Please try again.",
        };
      }

      return {
        success: true,
        message: "Signed out successfully.",
      };
    },
  }),
};

/**
 * Extracts a plain-text preview string from a block array.
 *
 * Walks the blocks in order and returns the content of the first
 * non-empty paragraph or heading block, trimmed to 200 characters.
 * Returns null if no suitable block is found (e.g. empty document).
 *
 * This preview is stored alongside the document and shown in the
 * sidebar — it must never contain markup or block syntax.
 *
 * @param blocks - The array of Block objects from the editor.
 * @returns A plain-text preview string, or null if the document is empty.
 */
function extractPreview(blocks: Block[]): string | null {
  for (const block of blocks) {
    if (
      (block.type === "paragraph" || block.type === "heading") &&
      block.content.trim().length > 0
    ) {
      return block.content.trim().slice(0, 200);
    }
  }
  return null;
}

/**
 * Zod schema for a single content block sent from the editor.
 *
 * Matches the Block interface in @/lib/journal/types.ts.
 * The meta field is an open record since different block types
 * store different metadata shapes.
 */
const blockSchema = z.object({
  id: z.string().min(1),
  type: z.enum([
    "paragraph",
    "heading",
    "quote",
    "separator",
    "checklist",
    "callout",
    "code",
    "prompt",
  ]),
  content: z.string(),
  meta: z.record(z.string(), z.unknown()).optional(),
});

/** A validated array of content blocks from the editor. */
const blocksSchema = z.array(blockSchema);

import { createClient } from "@/lib/supabase";
import { defineAction, ActionError } from "astro:actions";

/**
 * Retrieves the authenticated user from the current Supabase session.
 * Throws an UNAUTHORIZED ActionError if no session is found.
 *
 * @param context - The Astro action handler context.
 * @returns The authenticated Supabase user object and a ready supabase client.
 * @throws {ActionError} UNAUTHORIZED if the user is not logged in.
 */
export async function getAuthenticatedUser(
  context: Parameters<Parameters<typeof defineAction>[0]["handler"]>[1]
) {
  const supabase = createClient({
    request: context.request,
    cookies: context.cookies,
  });

  const {
    data: { user },
    error,
  } = await supabase.auth.getUser();

  if (error || !user) {
    throw new ActionError({
      code: "UNAUTHORIZED",
      message: "You must be logged in to perform this action.",
    });
  }

  return { supabase, user };
}
