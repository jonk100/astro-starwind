<script>
  import { actions } from "astro:actions";

  // ── Save status ────────────────────────────────────────────────────────────

  /**
   * Updates the save status indicator in the toolbar.
   *
   * The editor canvas dispatches a custom "editor:save-status" event
   * with a detail of "saving", "saved", or "error". This listener
   * reflects that state in the toolbar text and data-status attribute.
   *
   * @param status - The current save state.
   * @param message - Optional override message to display.
   */
  function setSaveStatus(status: "saving" | "saved" | "error", message?: string) {
    const el = document.getElementById("save-status");
    if (!el) return;

    const labels: Record<string, string> = {
      saving: "Saving…",
      saved: "Saved",
      error: "Save failed",
    };

    el.setAttribute("data-status", status);
    el.textContent = message ?? labels[status] ?? status;
  }

  window.addEventListener("editor:save-status", (e: Event) => {
    const detail = (e as CustomEvent<{ status: "saving" | "saved" | "error" }>).detail;
    setSaveStatus(detail.status);
  });

</script>

<script>
  import { actions } from "astro:actions";
  // ── Delete document ────────────────────────────────────────────────────────

  /**
   * Handles the delete button click.
   * Shows a native confirm dialog before calling the delete action.
   * On success, navigates back to the write app root (no open document).
   */
  function initDeleteButton() {
    const btn = document.getElementById("toolbar--delete-button") as HTMLButtonElement | null;
    if (!btn) return;

    const documentId = btn.getAttribute("data-document-id");
    if (!documentId) return;

    btn.addEventListener("click", async () => {
      const confirmed = window.confirm(
        "Delete this document? This cannot be undone."
      );
      if (!confirmed) return;

      btn.setAttribute("disabled", "true");

      const { error } = await actions.write.deleteDocument({ id: documentId });

      if (error) {
        btn.removeAttribute("disabled");
        console.error("Failed to delete document:", error.message);
        return;
      }

      // Navigate back to the write root with no document selected
      window.location.href = "/app/write";
    });
  }

    document.addEventListener("DOMContentLoaded", () => {
    initDeleteButton();
  });
</script>

<script>
  import { initPinStore, togglePin, pinStore } from "@/stores/journal/pin";

  /**
   * Syncs all UI attributes with store state.
   */
  function applyState(btn: HTMLButtonElement, pinned: boolean): void {
    btn.setAttribute("aria-pressed", pinned ? "true" : "false");
    btn.dataset.pinned = pinned ? "true" : "false";
    btn.setAttribute("aria-label", pinned ? "Unpin document" : "Pin document");
    btn.setAttribute("title", pinned ? "Unpin document" : "Pin document");
  }

  function initPinButton(): void {
    const btn = document.getElementById("toolbar--pin-button") as HTMLButtonElement | null;
    if (!btn) return;

    const documentId = btn.dataset.documentId;
    if (!documentId) return;

    const initialPinned = btn.dataset.pinned === "true";

    initPinStore(documentId, initialPinned);

    const unsubscribe = pinStore.subscribe((store) => {
      const state = store[documentId];
      if (!state) return;

      applyState(btn, state.pinned);
      btn.disabled = state.requesting;
    });

    btn.addEventListener("click", async () => {
      await togglePin(documentId);
    });

    window.addEventListener("beforeunload", unsubscribe);
  }

  document.addEventListener("DOMContentLoaded", initPinButton);
</script>

<script>
  const STORAGE_KEY = "write:toolbar-primary";
 
  interface BlockDef {
    type: string; icon: string; label: string; title: string; defaultPrimary: boolean;
  }
 
  function loadDefs(): BlockDef[] {
    const el = document.getElementById("bt-block-defs");
    try { return JSON.parse(el?.textContent ?? "[]") as BlockDef[]; } catch { return []; }
  }
 
  function loadPrimaryTypes(defs: BlockDef[]): string[] {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) return JSON.parse(saved) as string[];
    } catch { /* ignore */ }
    return defs.filter((d) => d.defaultPrimary).map((d) => d.type);
  }
 
  function savePrimaryTypes(types: string[]): void {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(types));
  }
 
  function initBlockToolbar(): void {
    const toolbar  = document.getElementById("block-toolbar");
    const primary  = document.getElementById("bt-primary");
    const moreBtnEl = document.getElementById("bt-more-trigger") as HTMLButtonElement | null;
    const dropdownEl = document.getElementById("bt-dropdown")    as HTMLUListElement  | null;
    const wrapper  = document.getElementById("bt-more-wrapper");
 
    if (!toolbar || !primary || !moreBtnEl || !dropdownEl || !wrapper) return;
 
    // Re-type as non-null for use inside closures
    const moreBtn:  HTMLButtonElement = moreBtnEl;
    const dropdown: HTMLUListElement  = dropdownEl;
 
    const defs = loadDefs();
    let primaryTypes = loadPrimaryTypes(defs);
    let lastActiveType: string | null = null;
 
    function dispatchBlockType(blockType: string): void {
      window.dispatchEvent(new CustomEvent("editor:set-block-type", { detail: { blockType } }));
    }
 
    function syncActiveType(activeType: string | null): void {
      lastActiveType = activeType;
      primary!.querySelectorAll<HTMLButtonElement>("[data-block-type]").forEach((btn) => {
        btn.setAttribute("aria-pressed", String(btn.dataset.blockType === activeType));
      });
      dropdown.querySelectorAll<HTMLElement>("[data-block-type]").forEach((opt) => {
        opt.setAttribute("aria-selected", String(opt.dataset.blockType === activeType));
      });
    }
 
    function renderPrimary(): void {
      primary!.innerHTML = "";
      primaryTypes.forEach((type, idx) => {
        const def = defs.find((d) => d.type === type);
        if (!def) return;
 
        const btn = document.createElement("button");
        btn.className           = "bt-btn";
        btn.type                = "button";
        btn.dataset.blockType   = type;
        btn.setAttribute("aria-label",   def.title);
        btn.setAttribute("aria-pressed", "false");
        btn.title               = `${def.title}${idx < 9 ? ` (Ctrl+${idx + 1})` : ""}`;
 
        const icon = document.createElement("span");
        icon.className   = "bt-icon";
        icon.textContent = def.icon;
        btn.appendChild(icon);
 
        if (idx < 9) {
          const badge = document.createElement("span");
          badge.className   = "bt-shortcut-badge";
          badge.textContent = String(idx + 1);
          btn.appendChild(badge);
        }
 
        btn.addEventListener("mousedown", (e) => e.preventDefault());
        btn.addEventListener("click", () => dispatchBlockType(type));
 
        // Right-click to demote
        btn.addEventListener("contextmenu", (e) => {
          e.preventDefault();
          primaryTypes = primaryTypes.filter((t) => t !== type);
          savePrimaryTypes(primaryTypes);
          renderPrimary();
          renderDropdown();
        });
 
        primary!.appendChild(btn);
      });
 
      // Hint shown at the bottom of primary area
      const hint = document.createElement("span");
      hint.className   = "bt-demote-hint";
      hint.style.display = "none"; // revealed inside dropdown instead
      syncActiveType(lastActiveType);
    }
 
    function renderDropdown(): void {
      dropdown.innerHTML = "";
      const secondaryDefs = defs.filter((d) => !primaryTypes.includes(d.type));
 
      if (secondaryDefs.length === 0) {
        const li = document.createElement("li");
        li.style.padding = "0.5rem 0.6rem";
        li.style.fontSize = "0.75rem";
        li.style.color = "var(--color-text-muted)";
        li.textContent = "All blocks are in the toolbar";
        dropdown.appendChild(li);
      } else {
        secondaryDefs.forEach((def) => {
          const li = document.createElement("li");
          li.className         = "bt-dropdown-option";
          li.dataset.blockType = def.type;
          li.setAttribute("role",          "option");
          li.setAttribute("aria-selected", "false");
 
          const iconEl = document.createElement("span");
          iconEl.className   = "bt-dropdown-icon";
          iconEl.textContent = def.icon;
 
          const titleEl = document.createElement("span");
          titleEl.className   = "bt-dropdown-title";
          titleEl.textContent = def.title;
 
          const pinBtn = document.createElement("button");
          pinBtn.className   = "bt-pin-btn";
          pinBtn.type        = "button";
          pinBtn.textContent = "Pin";
          pinBtn.title       = "Add to toolbar";
          pinBtn.addEventListener("mousedown", (e) => e.preventDefault());
          pinBtn.addEventListener("click", (e) => {
            e.stopPropagation();
            primaryTypes = primaryTypes.length < 9
              ? [...primaryTypes, def.type]
              : [...primaryTypes.slice(1), def.type];
            savePrimaryTypes(primaryTypes);
            renderPrimary();
            renderDropdown();
            dropdown.hidden = true;
            moreBtn.setAttribute("aria-expanded", "false");
          });
 
          li.appendChild(iconEl);
          li.appendChild(titleEl);
          li.appendChild(pinBtn);
          dropdown.appendChild(li);
        });
      }
 
      // Footer hint
      const hint = document.createElement("li");
      hint.style.cssText = "padding: 0.4rem 0.6rem; font-size: 0.65rem; color: var(--color-text-muted); border-top: 1px solid var(--color-border); margin-top: 0.25rem; list-style: none;";
      hint.textContent = "Right-click toolbar buttons to move here";
      dropdown.appendChild(hint);
    }
 
    // ── Dropdown open/close ────────────────────────────────────────────────
 
    moreBtn.addEventListener("mousedown", (e) => e.preventDefault());
    moreBtn.addEventListener("click", () => {
      const isOpen = !dropdown.hidden;
      dropdown.hidden = isOpen;
      moreBtn.setAttribute("aria-expanded", String(!isOpen));
    });
 
    dropdown.addEventListener("mousedown", (e) => e.preventDefault());
    dropdown.addEventListener("click", (e) => {
      const option = (e.target as HTMLElement).closest<HTMLElement>("[data-block-type]");
      if (!option || (e.target as HTMLElement).classList.contains("bt-pin-btn")) return;
      const blockType = option.dataset.blockType;
      if (blockType) dispatchBlockType(blockType);
      dropdown.hidden = true;
      moreBtn.setAttribute("aria-expanded", "false");
    });
 
    document.addEventListener("click", (e) => {
      if (!wrapper.contains(e.target as Node)) {
        dropdown.hidden = true;
        moreBtn.setAttribute("aria-expanded", "false");
      }
    });
 
    document.addEventListener("keydown", (e) => {
      if (e.key === "Escape") {
        dropdown.hidden = true;
        moreBtn.setAttribute("aria-expanded", "false");
      }
    });
 
    // ── Ctrl+N shortcuts ───────────────────────────────────────────────────
 
    document.addEventListener("keydown", (e) => {
      if (e.ctrlKey || e.metaKey) {
        toolbar.classList.add("ctrl-mode");
        const num = parseInt(e.key, 10);
        if (!isNaN(num) && num >= 1 && num <= primaryTypes.length) {
          e.preventDefault();
          dispatchBlockType(primaryTypes[num - 1]);
        }
      }
    });
 
    document.addEventListener("keyup", (e) => {
      if (e.key === "Control" || e.key === "Meta") {
        toolbar.classList.remove("ctrl-mode");
      }
    });
 
    // ── Focus sync ─────────────────────────────────────────────────────────
 
    window.addEventListener("editor:block-focused", ((e: CustomEvent) => {
      syncActiveType((e.detail as { blockType?: string }).blockType ?? null);
    }) as EventListener);
 
    window.addEventListener("editor:title-focused", () => syncActiveType(null));
 
    // ── Initial render ─────────────────────────────────────────────────────
    renderPrimary();
    renderDropdown();
  }
 
  document.addEventListener("DOMContentLoaded", () => {
    initBlockToolbar();
  });
</script>

<script>
  /**
   * Initialises the block type selector.
   */
  function initBlockTypeSelector(): void {
    const selector = document.getElementById("block-type-selector");

    const trigger =
      document.getElementById(
        "block-type-trigger"
      ) as HTMLButtonElement | null;

    const menu =
      document.getElementById(
        "block-type-menu"
      ) as HTMLUListElement | null;

    const label =
      document.getElementById(
        "block-type-label"
      ) as HTMLElement | null;

    if (!selector || !trigger || !menu || !label) {
      console.warn("[BlockTypeSelector] Missing required DOM nodes.");

      return;
    }

    const _selector = selector;
    const _trigger = trigger;
    const _menu = menu;
    const _label = label;

    /**
     * Human-readable labels keyed by block type.
     */
    const LABELS: Record<string, string> = {
      paragraph: "Paragraph",
      heading: "Heading",
      quote: "Quote",
      checklist: "Checklist",
      code: "Code",
      callout: "Callout",
      separator: "Divider",
    };

    /**
     * Prevent stale sync events briefly after selection.
     */
    let suppressSyncUntil = 0;

    /**
     * Opens the dropdown.
     */
    function openMenu(): void {
      console.log("[BlockTypeSelector] openMenu");

      _menu.hidden = false;

      _trigger.setAttribute("aria-expanded", "true");
    }

    /**
     * Closes the dropdown.
     */
    function closeMenu(): void {
      console.log("[BlockTypeSelector] closeMenu");

      _menu.hidden = true;

      _trigger.setAttribute("aria-expanded", "false");
    }

    /**
     * Toggle menu visibility.
     */
    _trigger.addEventListener("click", () => {
      console.log(
        "[BlockTypeSelector] trigger click",
        {
          menuHidden: _menu.hidden,
        }
      );

      _menu.hidden ? openMenu() : closeMenu();
    });

    /**
     * Close menu when clicking outside.
     */
    document.addEventListener("click", (event: MouseEvent) => {
      if (!_selector.contains(event.target as Node)) {
        console.log(
          "[BlockTypeSelector] outside click"
        );

        closeMenu();
      }
    });

    /**
     * Close on Escape.
     */
    document.addEventListener("keydown", (event: KeyboardEvent) => {
      if (event.key === "Escape") {
        console.log(
          "[BlockTypeSelector] escape pressed"
        );

        closeMenu();
      }
    });

    /**
     * Prevent dropdown options from stealing editor focus.
     */
    _menu.addEventListener("mousedown", (event: MouseEvent) => {
      console.log(
        "[BlockTypeSelector] prevent menu focus steal",
        {
          activeElement: document.activeElement,
        }
      );

      event.preventDefault();
    });

    /**
     * Handle block type selection.
     */
    _menu.addEventListener("click", (event: MouseEvent) => {
      const option = (
        event.target as HTMLElement
      ).closest<HTMLElement>("[data-block-type]");

      console.log(
        "[BlockTypeSelector] menu click",
        {
          target: event.target,
          option,
          activeElement: document.activeElement,
        }
      );

      if (!option) {
        console.warn(
          "[BlockTypeSelector] no option found"
        );

        return;
      }

      const blockType = option.getAttribute("data-block-type");

      if (!blockType) {
        console.warn(
          "[BlockTypeSelector] missing data-block-type"
        );

        return;
      }

      console.log(
        "[BlockTypeSelector] selected block type",
        {
          blockType,
          previousLabel: _label.textContent,
          activeElement: document.activeElement,
        }
      );

      suppressSyncUntil = Date.now() + 100;

      console.log(
        "[BlockTypeSelector] suppress sync until",
        suppressSyncUntil
      );

      _label.textContent = LABELS[blockType] ?? blockType;

      _menu
        .querySelectorAll<HTMLElement>("[aria-selected]")
        .forEach((element) => {
          element.setAttribute("aria-selected", "false");
        });

      option.setAttribute("aria-selected", "true");

      console.log(
        "[BlockTypeSelector] dispatch editor:set-block-type",
        {
          blockType,
        }
      );

      window.dispatchEvent(
        new CustomEvent("editor:set-block-type", {
          detail: { blockType },
        })
      );

      console.log(
        "[BlockTypeSelector] after dispatch",
        {
          activeElement: document.activeElement,
        }
      );

      closeMenu();
    });

    /**
     * Sync toolbar label with active editor block.
     */
    function syncLabel(blockType: string): void {
      console.log(
        "[BlockTypeSelector] syncLabel",
        {
          blockType,
          suppressSyncUntil,
          now: Date.now(),
          activeElement: document.activeElement,
        }
      );

      if (Date.now() < suppressSyncUntil) {
        console.warn(
          "[BlockTypeSelector] sync suppressed"
        );

        return;
      }

      _label.textContent = LABELS[blockType] ?? "Paragraph";

      _menu
        .querySelectorAll<HTMLElement>("[data-block-type]")
        .forEach((element) => {
          element.setAttribute(
            "aria-selected",
            String(
              element.getAttribute("data-block-type") ===
                blockType
            )
          );
        });

      console.log(
        "[BlockTypeSelector] sync complete",
        {
          finalLabel: _label.textContent,
        }
      );
    }

    /**
     * Sync when editor block focus changes.
     */
    window.addEventListener(
      "editor:block-focused",
      ((event: CustomEvent) => {
        console.log(
          "[BlockTypeSelector] editor:block-focused",
          {
            detail: event.detail,
            activeElement: document.activeElement,
          }
        );

        syncLabel(
          (event.detail as { blockType?: string })
            .blockType ?? "paragraph"
        );
      }) as EventListener
    );

    /**
     * Clear label when title receives focus.
     */
    window.addEventListener("editor:title-focused", () => {
      console.log(
        "[BlockTypeSelector] editor:title-focused"
      );

      _label.textContent = "—";
    });

    console.log(
      "[BlockTypeSelector] initialised"
    );
  }

  document.addEventListener("DOMContentLoaded", () => {
    console.log(
      "[BlockTypeSelector] DOMContentLoaded"
    );

    initBlockTypeSelector();
  });
</script>
