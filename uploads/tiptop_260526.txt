/**
 * @file src/components/apps/tiptop/TipTop.astro
 * @description TipTap-powered editor component for the Write app.
 *
 * Replaces EditorCanvas.astro. Wires TipTap directly into the existing
 * XState machine and editorStore without changing either — the only new
 * surface is the tiptap.ts translation layer.
 *
 * Data flow:
 *   1. LOAD:  blocksToTipTap(activeBlocks) → editor.commands.setContent()
 *   2. EDIT:  editor.onUpdate → tiptapToBlocks() → machine.send(BLOCKS_CHANGED)
 *   3. TITLE: title input → machine.send(TITLE_CHANGED) + updateTitle()
 *   4. SAVE:  XState machine → actions.write.saveBlocks (Block[] format, unchanged)
 *
 * The `documentId` and `title` props are used to seed the XState machine
 * on first mount. If no documentId is provided (empty state), the editor
 * renders a placeholder and disables saving.
 */

interface Props {
  /** The UUID of the currently open document, or empty string if none. */
  documentId: string;
  /** The current document title. */
  title: string;
}
import './TipTop.css';
import '../write/editor/editorCanvas.css';

const { documentId, title } = Astro.props;

<script>
  import { Editor } from '@tiptap/core';
  import StarterKit from '@tiptap/starter-kit';
  import { TaskList } from '@tiptap/extension-task-list';
  import { TaskItem } from '@tiptap/extension-task-item';
  import { Placeholder } from '@tiptap/extension-placeholder';

  import { createActor } from 'xstate';
  import { editorMachine } from '@/components/apps/write/state/editorMachine';
  import { setActiveDocument, updateTitle } from '@/stores/editor';
  import { tiptapToBlocks, blocksToTipTap } from '@/lib/journal/tiptap';
  import type { Block } from '@/lib/journal/types';

  // ─── Initialization ──────────────────────────────────────────────────────

  /**
   * Bootstraps the TipTap editor and wires it to the XState machine.
   *
   * Reads `documentId` and initial blocks from the DOM (injected by the
   * Astro page into script tags to avoid SSR import issues). Starts the
   * XState machine actor, then creates the TipTap editor instance.
   */
  function initTipTop(): void {
    const shell = document.querySelector<HTMLElement>('.tiptop-shell');
    if (!shell) return;

    const documentId = shell.dataset.documentId ?? '';
    const title      = shell.dataset.title ?? '';

    // Blocks are injected by index.astro as JSON in a script tag
    const blocksEl  = document.getElementById('tiptap-content');
    let initialBlocks: Block[] = [];

    if (blocksEl?.textContent) {
      try {
        const parsed = JSON.parse(blocksEl.textContent);
        if (Array.isArray(parsed)) {
          initialBlocks = parsed as Block[];
        }
      } catch (e) {
        console.error('[TipTop] Failed to parse initial blocks:', e);
      }
    }

    // ── Seed editorStore ────────────────────────────────────────────────────
    // This keeps the nanostores-based computed values (wordCount, preview)
    // in sync from the first render.
    if (documentId) {
      setActiveDocument(documentId, title, initialBlocks);
    }

    // ── Start XState machine ────────────────────────────────────────────────
    // The machine manages debounced autosave. It expects the same context
    // shape as before — no changes needed to editorMachine.ts.
    const machine = documentId
      ? createActor(editorMachine, {
          input: {
            documentId,
            title,
            blocks: initialBlocks,
          },
        })
      : null;

    machine?.start();

    // ── Create TipTap editor ────────────────────────────────────────────────
    const editorEl = document.getElementById('tiptop-editor');
    if (!editorEl) return;

    /**
     * Convert the stored Block[] into TipTap's JSON format so the
     * existing content appears correctly on load.
     */
    const initialContent = blocksToTipTap(initialBlocks);

    const editor = new Editor({
      element: editorEl,

      extensions: [
        StarterKit,
        TaskList,
        TaskItem.configure({ nested: false }),
        Placeholder.configure({
          placeholder: 'Write something…',
        }),
      ],

      content: initialContent,

      editable: Boolean(documentId),

      /**
       * Called on every content change (keystroke, paste, format, etc.).
       *
       * Converts the TipTap JSON to Block[] using the translation layer,
       * then sends a BLOCKS_CHANGED event to the XState machine so the
       * existing debounced autosave cycle kicks in.
       */
      onUpdate({ editor }) {
        if (!machine) return;

        const doc    = editor.getJSON() as Parameters<typeof tiptapToBlocks>[0];
        const blocks = tiptapToBlocks(doc);

        machine.send({ type: 'BLOCKS_CHANGED', blocks });

        // Also update editorStore so nanostores computed values stay current
        // (wordCount, documentPreview) without re-reading the DOM.
        import('@/stores/editor').then(({ updateBlocks }) => {
          updateBlocks(blocks);
        });

        updateWordCount(editor.getText());
      },
    });

    // ── Title field ─────────────────────────────────────────────────────────
    initTitleField(machine);

    // ── Toolbar integration ─────────────────────────────────────────────────
    // The existing BlockToolbar dispatches 'editor:set-block-type' events.
    // Map those to TipTap commands so the toolbar still works.
    if (documentId) {
      initToolbarBridge(editor);
    }

    // ── Cleanup ─────────────────────────────────────────────────────────────
    window.addEventListener('beforeunload', () => {
      editor.destroy();
      machine?.stop();
    });
  }

  // ─── Title field ──────────────────────────────────────────────────────────

  /**
   * Wires the contenteditable title element to the XState machine.
   *
   * Uses the existing `TITLE_CHANGED` event — no changes to the machine
   * are required. Also calls `updateTitle()` to keep the editorStore
   * and sidebar link text in sync.
   *
   * @param machine - The running XState actor, or null if no doc is open.
   */
  function initTitleField(
    machine: ReturnType<typeof createActor<typeof editorMachine>> | null
  ): void {
    const titleEl = document.getElementById('tiptop-title') as HTMLElement | null;
    if (!titleEl || !machine) return;

    titleEl.addEventListener('input', () => {
      const newTitle = titleEl.textContent?.trim() ?? '';
      machine.send({ type: 'TITLE_CHANGED', title: newTitle });
      updateTitle(newTitle);
    });

    // Prevent Enter from inserting a newline in the title field
    titleEl.addEventListener('keydown', (e: KeyboardEvent) => {
      if (e.key === 'Enter') {
        e.preventDefault();
        // Move focus to the editor body
        document.getElementById('tiptop-editor')?.querySelector<HTMLElement>('.ProseMirror')?.focus();
      }
    });

    // Announce title focus to the BlockTypeSelector (matches old EditorCanvas behaviour)
    titleEl.addEventListener('focus', () => {
      window.dispatchEvent(new CustomEvent('editor:title-focused'));
    });
  }

  // ─── Toolbar bridge ───────────────────────────────────────────────────────

  /**
   * Maps 'editor:set-block-type' custom events from the existing BlockToolbar
   * to the equivalent TipTap commands.
   *
   * The toolbar was built for the old contenteditable block editor and
   * dispatches string block types. This bridge translates them to the
   * TipTap command API so the toolbar keeps working without modification.
   *
   * @param editor - The live TipTap Editor instance.
   */
  function initToolbarBridge(editor: Editor): void {
    window.addEventListener('editor:set-block-type', (e: Event) => {
      const detail = (e as CustomEvent<{ blockType: string }>).detail;

      // Focus the editor first so commands apply to the cursor position
      editor.commands.focus();

      switch (detail.blockType) {
        case 'paragraph':
          editor.chain().focus().setParagraph().run();
          break;

        case 'heading':
        case 'heading-1':
          editor.chain().focus().setHeading({ level: 1 }).run();
          break;
        case 'heading-2':
          editor.chain().focus().setHeading({ level: 2 }).run();
          break;
        case 'heading-3':
          editor.chain().focus().setHeading({ level: 3 }).run();
          break;
        case 'heading-4':
          editor.chain().focus().setHeading({ level: 4 }).run();
          break;
        case 'heading-5':
          editor.chain().focus().setHeading({ level: 5 }).run();
          break;

        case 'quote':
          editor.chain().focus().toggleBlockquote().run();
          break;

        case 'code':
          editor.chain().focus().toggleCodeBlock().run();
          break;

        case 'separator':
          editor.chain().focus().setHorizontalRule().run();
          break;

        case 'checklist':
          editor.chain().focus().toggleTaskList().run();
          break;

        // prompt and callouts have no direct TipTap equivalent yet —
        // fall back to a plain paragraph for now.
        case 'prompt':
        case 'callout':
        case 'callout-info':
        case 'callout-warning':
        case 'callout-success':
        case 'callout-danger':
          editor.chain().focus().setParagraph().run();
          break;
      }
    });
  }

  // ─── Word count ───────────────────────────────────────────────────────────

  /**
   * Updates the word count display in the footer.
   *
   * @param plainText - The plain text extracted by editor.getText().
   */
  function updateWordCount(plainText: string): void {
    const countEl = document.getElementById('tiptop-word-count');
    if (!countEl) return;

    const count = plainText.trim().split(/\s+/).filter(Boolean).length;
    countEl.textContent = `${count} ${count === 1 ? 'word' : 'words'}`;
  }

  // ─── Boot ─────────────────────────────────────────────────────────────────

  document.addEventListener('DOMContentLoaded', initTipTop);
</script>

import type { Block, TiptapNode, TiptapDoc } from "./tiptap.types";

// ── Block[] → Tiptap HTML (load) ─────────────────────────────────────────

/**
 * Converts your Block[] from the database into an HTML string that
 * Tiptap can render on initialisation.
 *
 * We generate HTML rather than Tiptap JSON because it's simpler to
 * produce and Tiptap's `content` option accepts either.
 *
 * Each block type maps to the HTML element StarterKit understands:
 * - paragraph   → <p>
 * - heading     → <h1> (meta.level drives h1-h5)
 * - heading-1   → <h1>
 * - heading-2   → <h2>
 * - heading-3   → <h3>
 * - heading-4   → <h4>
 * - heading-5   → <h5>
 * - quote       → <blockquote><p>
 * - checklist   → <ul><li>  (StarterKit bulletList)
 * - code        → <pre><code>
 * - separator   → <hr>
 * - callout     → <p>  (no native Tiptap callout; degrades gracefully)
 * - prompt      → <p>
 *
 * Content is escaped so stray < > & characters don't break the HTML.
 *
 * @param blocks - The Block[] array from the database.
 * @returns An HTML string suitable for Tiptap's `content` option.
 */
export function blocksToHtml(blocks: Block[]): string {
  /**
   * Escapes the five XML/HTML special characters in a plain-text string.
   * Required because block content is user-supplied and may contain < > & " '
   */
  function esc(str: string): string {
    return str
      .replace(/&/g,  "&amp;")
      .replace(/</g,  "&lt;")
      .replace(/>/g,  "&gt;")
      .replace(/"/g,  "&quot;")
      .replace(/'/g,  "&#39;");
  }

  return blocks.map((block) => {
    const text = esc(block.content ?? "");

    switch (block.type) {
      case "heading": {
        // The old editor stored level in meta, defaulting to 1
        const level = (block.meta?.level as number) ?? 1;
        return `<h${level}>${text}</h${level}>`;
      }
      case "heading-1": return `<h1>${text}</h1>`;
      case "heading-2": return `<h2>${text}</h2>`;
      case "heading-3": return `<h3>${text}</h3>`;
      case "heading-4": return `<h4>${text}</h4>`;
      case "heading-5": return `<h5>${text}</h5>`;
      case "quote":     return `<blockquote><p>${text}</p></blockquote>`;
      case "checklist": return `<ul><li>${text}</li></ul>`;
      case "code":      return `<pre><code>${text}</code></pre>`;
      case "separator": return `<hr>`;
      // callout, prompt, and any unknown types degrade to paragraph
      default:          return `<p>${text}</p>`;
    }
  }).join("\n");
}

// ── Tiptap JSON → Block[] (save) ──────────────────────────────────────────

/**
 * Extracts the plain-text content from a Tiptap inline node array.
 * Walks the child nodes and concatenates any text nodes, ignoring marks
 * (bold, italic etc.) since Block.content is plain text only.
 *
 * @param children - The `content` array of a Tiptap block node.
 * @returns A plain-text string.
 */
export function inlineText(children: TiptapNode[] = []): string {
  return children
    .filter((n) => n.type === "text")
    .map((n)    => n.text ?? "")
    .join("");
}

/**
 * Converts a Tiptap JSON document back to Block[] so the existing
 * saveBlocks action and blockSchema validation need no changes.
 *
 * Tiptap node types → Block types:
 * - paragraph      → "paragraph"
 * - heading (l1)   → "heading"   (meta.level = 1, for legacy compat)
 * - heading (l2-5) → "heading-2" … "heading-5"
 * - blockquote     → "quote"
 * - bulletList     → "checklist" (first listItem's text only)
 * - orderedList    → "checklist" (treated same — no ordered list Block type)
 * - codeBlock      → "code"
 * - horizontalRule → "separator"
 * - unknown        → "paragraph"
 *
 * Each block gets a fresh crypto.randomUUID() because Tiptap has no concept
 * of stable block IDs — the IDs are only needed for the DB row.
 *
 * @param doc - The Tiptap JSON document from editor.getJSON().
 * @returns A Block[] array matching the blockSchema in write_actions.ts.
 */
export function tiptapDocToBlocks(doc: TiptapDoc): Block[] {
  return (doc.content ?? []).map((node): Block => {
    const id = crypto.randomUUID();
 
    switch (node.type) {
      case "heading": {
        const level = (node.attrs?.level as number) ?? 1;
        // H1 maps to "heading" (legacy type) so old documents stay compatible
        const type  = level === 1 ? "heading" : `heading-${level}`;
        return { id, type, content: inlineText(node.content), meta: { level } };
      }
 
      case "blockquote": {
        // blockquote wraps one or more paragraph nodes — flatten to one string
        const text = (node.content ?? [])
          .map((child) => inlineText(child.content))
          .join("\n");
        return { id, type: "quote", content: text };
      }
 
      case "bulletList":
      case "orderedList": {
        // Flatten all list items into one block separated by newlines.
        // There is no multi-item Block type, so this is the closest mapping.
        const text = (node.content ?? [])
          .map((li) =>
            (li.content ?? [])
              .map((child) => inlineText(child.content))
              .join(" ")
          )
          .join("\n");
        return { id, type: "checklist", content: text };
      }
 
      case "codeBlock":
        return { id, type: "code",      content: inlineText(node.content) };
 
      case "horizontalRule":
        return { id, type: "separator", content: "" };
 
      default:
        return { id, type: "paragraph", content: inlineText(node.content) };
    }
  });
}

// ── Preview extraction ───────────────────────────────────────────────────

/**
 * Extracts a sidebar preview string from a Block[].
 * Mirrors extractPreview() in write_actions.ts — returns the text of the
 * first non-empty paragraph or heading block, trimmed to 200 characters.
 *
 * We run this client-side so we don't have to send the full block array
 * just to compute a preview on the server.
 *
 * @param blocks - The converted Block[] about to be saved.
 * @returns A plain-text preview string, or null if the document is empty.
 */
export function extractPreview(blocks: Block[]): string | null {
  for (const block of blocks) {
    if (
      (block.type === "paragraph" || block.type === "heading") &&
      block.content.trim().length > 0
    ) {
      return block.content.trim().slice(0, 200);
    }
  }
  return null;
}

  function initColorPickers(): void {
    const textColorPicker = document.getElementById('tb-text-color');
    const highlightPicker = document.getElementById('tb-highlight-color');

    textColorPicker?.addEventListener('starwind-color-picker:change', (e: Event) => {
      const { value } = (e as ColorPickerChangeEvent).detail;
      dispatch('setColor', { color: value });
    });

    highlightPicker?.addEventListener('starwind-color-picker:change', (e: Event) => {
      const { value } = (e as ColorPickerChangeEvent).detail;
      dispatch('setHighlight', { color: value });
    });
  }

  // ─── Emoji ─────────────────────────────────────────────────────────────────

  /**
   * Triggers TipTap's emoji extension suggestion by dispatching an
   * 'insertEmoji' command, which the editor bridge handles.
   */
  function initEmojiButton(): void {
    const btn = document.getElementById('tb-emoji') as HTMLButtonElement | null;
    if (!btn) return;

    btn.addEventListener('click', () => dispatch('insertEmoji'));
  }

  // ─── Undo / redo ───────────────────────────────────────────────────────────

  /**
   * Wires undo and redo buttons.
   */
  function initHistory(): void {
    document.getElementById('tb-undo')?.addEventListener('click', () => dispatch('undo'));
    document.getElementById('tb-redo')?.addEventListener('click', () => dispatch('redo'));
  }

  // ─── Theme toggle ──────────────────────────────────────────────────────────

  /**
   * Toggles dark/light mode by flipping `data-theme` on <html> and
   * persisting the choice to localStorage.
   *
   * Assumes your global CSS uses `[data-theme="dark"]` / `[data-theme="light"]`
   * or the `.dark` class — swap the selector below to match your setup.
   */
  function initThemeToggle(): void {
    const btn       = document.getElementById('tb-theme') as HTMLButtonElement | null;
    const iconDark  = document.getElementById('tb-theme-icon-dark')  as SVGSVGElement | null;
    const iconLight = document.getElementById('tb-theme-icon-light') as SVGSVGElement | null;
    if (!btn) return;

    /** Returns true if dark mode is currently active. */
    function isDark(): boolean {
      return document.documentElement.getAttribute('data-theme') === 'dark'
        || document.documentElement.classList.contains('dark');
    }

    /** Syncs the button icon to the current theme. */
    function syncIcon(): void {
      const dark = isDark();
      if (iconDark)  iconDark.style.display  = dark ? 'block' : 'none';
      if (iconLight) iconLight.style.display = dark ? 'none'  : 'block';
    }

    syncIcon();

    btn.addEventListener('click', () => {
      const next = isDark() ? 'light' : 'dark';

      document.documentElement.setAttribute('data-theme', next);
      document.documentElement.classList.toggle('dark', next === 'dark');
      localStorage.setItem('theme', next);
      syncIcon();

      window.dispatchEvent(new CustomEvent('theme:toggle', { detail: { theme: next } }));
    });
  }

  // ─── Boot ──────────────────────────────────────────────────────────────────

  document.addEventListener('DOMContentLoaded', () => {
    initBlockTypeDropdown();
    initFontFamilyDropdown();
    initFontSizeDropdown();
    initMarkButtons();
    initListButtons();
    initColorPickers();
    initEmojiButton();
    initHistory();
    initThemeToggle();
    // The shortcuts panel is now a Starwind Dropdown — no custom JS needed.
  });
</script>
